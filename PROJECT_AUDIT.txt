# ai_analytics/detector.py
import math
import cv2
from ultralytics import YOLO
from logging_module.logger import get_logger

logger = get_logger("ai_detector")

class TargetDetector:
    def __init__(self, confidence_threshold: float = 0.4):
        self.confidence_threshold = confidence_threshold
        # Load lightweight nano model (downloads automatically on first run)
        self.model = YOLO("yolov8n.pt")

    def process_frame(self, frame_bgr, frame_metadata: dict, telemetry: dict) -> list:
        """
        Runs real YOLO model on the live frame and calculates geo-coordinates.
        """
        results = self.model(frame_bgr, verbose=False)[0]
        detections = []

        for box in results.boxes:
            conf = float(box.conf[0])
            if conf >= self.confidence_threshold:
                cls_id = int(box.cls[0])
                class_name = self.model.names[cls_id]
                xyxy = box.xyxy[0].cpu().numpy().astype(int).tolist()

                target_geo = self._calculate_target_coordinates(
                    drone_lat=telemetry.get("lat", 0.0),
                    drone_lon=telemetry.get("lon", 0.0),
                    alt=telemetry.get("alt", 50.0),
                    heading=telemetry.get("heading", 0.0)
                )

                detections.append({
                    "class_name": class_name,
                    "confidence": round(conf, 2),
                    "bbox": xyxy,
                    "latitude": target_geo["lat"],
                    "longitude": target_geo["lon"]
                })

        return detections

    def _calculate_target_coordinates(self, drone_lat: float, drone_lon: float, alt: float, heading: float) -> dict:
        earth_radius = 6378137.0
        offset_meters = alt * math.tan(math.radians(15))
        bearing = math.radians(heading)
        
        d_lat = (offset_meters * math.cos(bearing)) / earth_radius
        d_lon = (offset_meters * math.sin(bearing)) / (earth_radius * math.cos(math.radians(drone_lat)))

        return {
            "lat": drone_lat + math.degrees(d_lat),
            "lon": drone_lon + math.degrees(d_lon)
        }

import structlog

RABBITMQ_QUEUE = "alerts_queue"


async def process_alert(message: aio_pika.IncomingMessage):
    async with message.process():
        alert_data = json.loads(message.body.decode())
        
        # Renamed keyword parameter from event to alert_event to avoid structlog key collision
        logger.warning(
            "🚨 CRITICAL ALERT DISPATCHED 🚨",
            alert_event=alert_data.get("event"),
            target=alert_data.get("class"),
            confidence=f"{int(alert_data.get('confidence', 0) * 100)}%",
            lat=alert_data.get("latitude"),
            lon=alert_data.get("longitude")
        )
        
        target_name = alert_data.get("class", "UNKNOWN").upper()
        lat = alert_data.get("latitude")
        lon = alert_data.get("longitude")
        print(f"\n[ALERT WORKER] -> ACTION REQUIRED: High confidence {target_name} identified at LAT: {lat}, LON: {lon}\n")


async def main():
    logger.info("starting_rabbitmq_alert_consumer_worker")
    connection = await aio_pika.connect_robust(RABBITMQ_URL)
    
    async with connection:
        channel = await connection.channel()
        await channel.set_qos(prefetch_count=1)
        queue = await channel.declare_queue(RABBITMQ_QUEUE, durable=True)
        
        logger.info("listening_for_threat_alerts_on_rabbitmq", queue=RABBITMQ_QUEUE)
        
        await queue.consume(process_alert)
        await asyncio.Future()

    """Dynamically fetches your PC's local network IP."""
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:

        ip = s.getsockname()[0]
    except Exception:

    finally:
        s.close()
    return ip

async def websocket_telemetry(websocket: WebSocket):
    """
    Bi-directional communication channel:
    - Receives telemetry targets from reader scripts and routes them to C2 UI
    - Receives Admin config commands from C2 UI and broadcasts them
    """
    await manager.connect(websocket)
    try:
        while True:
            data_str = await websocket.receive_text()
            data = json.loads(data_str)

            # Route 1: Target Detection Payload
            if data.get("type") == "detection":
                targets = data.get("targets", [])
                
                # Save to DB if PostGIS is online
                if DB_ONLINE and targets:
                    async with AsyncSessionLocal() as session:
                        for t in targets:
                            log = DetectionLog(
                                class_name=t.get("class_name", "unknown"),
                                confidence=t.get("confidence", 0.0),
                                latitude=t.get("latitude", 0.0),
                                longitude=t.get("longitude", 0.0),
                                engine=t.get("engine", "C2")
                            )
                            session.add(log)
                        await session.commit()

                # Broadcast live detections to all connected C2 Dashboards
                await manager.broadcast(data_str)

            # Route 2: Admin Config Overrides from Frontend
            elif data.get("type") == "admin_config":
                print(f"[ADMIN C2 OVERRIDE] Broadcasted engine parameters: {data}")
                await manager.broadcast(data_str)

    except WebSocketDisconnect:
        manager.disconnect(websocket)
    except Exception as e:
        print(f"[SOCKET ERROR] {e}")
        manager.disconnect(websocket)

# auth_module/auth.py
from datetime import datetime, timedelta, timezone
import jwt
from fastapi import HTTPException, Security, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from config_module.settings import settings

security = HTTPBearer()

def create_access_token(data: dict, expires_delta: timedelta | None = None) -> str:
    to_encode = data.copy()
    expire = datetime.now(timezone.utc) + (expires_delta or timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES))
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)

def decode_token(token: str) -> dict:
    try:
        return jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
    except jwt.PyJWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials",
            headers={"WWW-Authenticate": "Bearer"},
        )

def require_role(required_role: str):
    def role_dependency(credentials: HTTPAuthorizationCredentials = Security(security)):
        payload = decode_token(credentials.credentials)
        user_role = payload.get("role")
        if user_role != required_role and user_role != "admin":
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Insufficient permissions for this operation",
            )
        return payload
    return role_dependency

from sqlalchemy.ext.asyncio import AsyncSession
from geoalchemy2.shape import from_shape
from shapely.geometry import Point
from database_module.models import DetectionLog
from database_module.session import AsyncSessionLocal
from logging_module.logger import get_logger

logger = get_logger("detection_repository")

async def log_detections_to_db(stream_id: str, detections: list):
    """
    Asynchronously persists target detection events into PostGIS.
    """
    if not detections:
        return

    async with AsyncSessionLocal() as session:
        try:
            for det in detections:
                lat = det.get("latitude")
                lon = det.get("longitude")
                
                # Create Spatial Point geometry for PostGIS
                point = Point(lon, lat)
                spatial_geom = from_shape(point, srid=4326)

                log_entry = DetectionLog(
                    stream_id=stream_id,
                    class_name=det.get("class_name"),
                    confidence=det.get("confidence"),
                    latitude=lat,
                    longitude=lon,
                    location=spatial_geom
                )
                session.add(log_entry)

            await session.commit()
            logger.info("detections_logged_to_postgis", count=len(detections))
        except Exception as e:
            await session.rollback()
            logger.error("failed_to_log_detections", error=str(e))

def get_class_color(class_name: str):
    """Generates a deterministic, high-contrast BGR color based on class string."""
    hash_val = zlib.crc32(class_name.lower().encode('utf-8'))
    b = (hash_val & 0xFF)
    g = ((hash_val >> 8) & 0xFF)
    r = ((hash_val >> 16) & 0xFF)
    
    max_val = max(r, g, b)
    if max_val < 130:
        r = min(255, r + 100)
        g = min(255, g + 100)
        b = min(255, b + 100)
        
    return (int(b), int(g), int(r))


def setup_display_window(window_name=WINDOW_NAME):
    """Configures a resizable OpenCV GUI window that retains aspect ratio."""
    cv2.namedWindow(window_name, cv2.WINDOW_NORMAL | cv2.WINDOW_KEEPRATIO)


def get_frame_orientation(frame):
    """Calculates live frame dimensions, orientation mode, and aspect ratio."""
    h, w = frame.shape[0], frame.shape[1]
    orientation = "portrait" if h > w else "landscape"
    ratio = f"{w}:{h}"
    return {"width": w, "height": h, "orientation": orientation, "aspect_ratio": ratio}


def resize_for_display(frame, target_height=720):
    """Scales display frame height cleanly without breaking aspect ratio."""
    h, w = frame.shape[0], frame.shape[1]
    if h == 0 or w == 0:
        return frame
    scale = target_height / float(h)
    new_w = int(w * scale)
    return cv2.resize(frame, (new_w, target_height), interpolation=cv2.INTER_LINEAR)


def apply_clahe(frame):
    """Applies local CLAHE contrast enhancement for dark or high-exposure streams."""
    lab = cv2.cvtColor(frame, cv2.COLOR_BGR2LAB)
    l, a, b = cv2.split(lab)
    clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8))
    cl = clahe.apply(l)
    return cv2.cvtColor(cv2.merge((cl, a, b)), cv2.COLOR_LAB2BGR)


def non_max_suppression_fast(boxes, overlap_thresh=0.45):
    """Combines predictions across FAST and BROAD AI engines using standard NMS."""
    if not boxes:
        return []
    arr = np.array([[b['x1'], b['y1'], b['x2'], b['y2'], b['confidence']] for b in boxes])
    x1, y1, x2, y2, scores = arr[:, 0], arr[:, 1], arr[:, 2], arr[:, 3], arr[:, 4]
    area = (x2 - x1 + 1) * (y2 - y1 + 1)
    idxs = np.argsort(scores)
    pick = []
    
    while len(idxs) > 0:
        last = len(idxs) - 1
        i = idxs[last]
        pick.append(i)
        xx1 = np.maximum(x1[i], x1[idxs[:last]])
        yy1 = np.maximum(y1[i], y1[idxs[:last]])
        xx2 = np.minimum(x2[i], x2[idxs[:last]])
        yy2 = np.minimum(y2[i], y2[idxs[:last]])
        w = np.maximum(0, xx2 - xx1 + 1)
        h = np.maximum(0, yy2 - yy1 + 1)
        overlap = (w * h) / area[idxs[:last]]
        idxs = np.delete(idxs, np.concatenate(([last], np.where(overlap > overlap_thresh)[0])))
        
    return [boxes[p] for p in pick]


def run_fast_engine(model_fast, frame, min_conf):
    """Executes primary COCO object detection model."""
    dets = []
    res = model_fast(frame, verbose=False)[0]
    for box in res.boxes:
        conf = float(box.conf[0])
        if conf >= min_conf:
            x1, y1, x2, y2 = map(int, box.xyxy[0])
            cls_name = model_fast.names[int(box.cls[0])].lower()
            dets.append({
                "class_name": cls_name, "confidence": conf,
                "x1": x1, "y1": y1, "x2": x2, "y2": y2, "engine": "FAST"
            })
    return dets

    thickness = max(2, int(font_scale * 2.5))
    
    for item in final_targets:
        x1, y1, x2, y2 = item['x1'], item['y1'], item['x2'], item['y2']
        cls_name = item['class_name'].upper()
        conf = item['confidence']
        color = get_class_color(item['class_name'])
        
        # Target Bounding Box
        cv2.rectangle(frame, (x1, y1), (x2, y2), color, thickness + 1)
        
        # Label Typography Backdrop
        label_text = f"{cls_name} {conf:.2f}"
        (text_w, text_h), _ = cv2.getTextSize(label_text, cv2.FONT_HERSHEY_SIMPLEX, font_scale, thickness)
        
        label_y1 = max(y1 - text_h - 10, 0)
        label_y2 = max(y1, text_h + 10)
        cv2.rectangle(frame, (x1, label_y1), (x1 + text_w + 10, label_y2), color, -1)

        telemetry_payload.append({
            "class_name": item['class_name'],
            "confidence": round(conf, 2),
            "latitude": BASE_LAT,
            "longitude": BASE_LON,
            "engine": item['engine']
        })
        
    return telemetry_payload

class NonBlockingRTMPReader:
    """Thread-safe frame buffer that reads continuously from MediaMTX."""
    def __init__(self, rtmp_src):
        self.rtmp_src = rtmp_src
        self.cap = None
        self.q = Queue(maxsize=2)
        self.running = True
        self.connected = False
        self.thread = threading.Thread(target=self._update, daemon=True)
        self.thread.start()

    def _connect(self):
        print(f"[INFO] Connecting to RTMP stream: {self.rtmp_src}")
        while self.running:
            self.cap = cv2.VideoCapture(self.rtmp_src, cv2.CAP_FFMPEG)
            self.cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)
            if self.cap.isOpened():
                ret, _ = self.cap.read()
                if ret:
                    print("[SUCCESS] Live RTMP video feed acquired.")
                    self.connected = True
                    return
            if self.cap:
                self.cap.release()
            print("[WAITING] MediaMTX stream offline. Retrying in 2 seconds...")
            time.sleep(2)

    def _update(self):
        self._connect()
        while self.running:
            if not self.cap or not self.cap.isOpened():
                self.connected = False
                self._connect()
                continue
            ret, frame = self.cap.read()
            if not ret:
                print("[WARN] Video stream dropped. Reconnecting...")
                self.connected = False
                self.cap.release()
                time.sleep(1)
                self._connect()
                continue
            
            # Clear old queue elements to enforce zero frame buffer lag
            if not self.q.empty():
                try:
                    self.q.get_nowait()
                except Exception:
                    pass
            self.q.put(frame)

    def read(self):
        return self.q.get() if not self.q.empty() else None

    def stop(self):
        self.running = False
        if self.cap and self.cap.isOpened():
            self.cap.release()


async def receive_admin_commands(websocket):
    """Receives dynamically pushed C2 control parameters."""
    global ADMIN_CONFIG
    try:
        async for message in websocket:
            data = json.loads(message)
            if data.get("type") == "admin_config":
                for k in ("engine_mode", "min_confidence", "max_area_pct", "clahe_enabled"):
                    if k in data:
                        ADMIN_CONFIG[k] = data[k]
                print(f"[C2 CONFIG UPDATE] -> {ADMIN_CONFIG}")
    except Exception:
        pass


async def run_pipeline():
    reader = NonBlockingRTMPReader(RTMP_URL)
    frame_count = 0
    last_orientation = None

    window_title = "DECS - Live Target Tracking (RTMP)"
    setup_display_window(window_title)

    websocket = None
    # Resilient connection to C2 WebSocket Telemetry Server
    try:
        websocket = await asyncio.wait_for(websockets.connect(WS_URL), timeout=2.0)
        print(f"[INFO] Telemetry pipeline connected to server: {WS_URL}")
        asyncio.create_task(receive_admin_commands(websocket))
    except Exception as ws_err:
        print(f"[WARN] C2 Telemetry Server offline ({ws_err}). Pipeline running in local display mode.")

    try:
        while True:
            frame = reader.read()
            if frame is None:
                await asyncio.sleep(0.005)
                continue

            frame_count += 1
            mode = ADMIN_CONFIG["engine_mode"]
            min_conf = ADMIN_CONFIG["min_confidence"]
            max_area_px = (ADMIN_CONFIG["max_area_pct"] / 100.0) * (frame.shape[0] * frame.shape[1])
            working_frame = apply_clahe(frame) if ADMIN_CONFIG["clahe_enabled"] else frame

            # Monitor dynamic video aspect ratios
            frame_meta = get_frame_orientation(frame)
            if frame_meta["orientation"] != last_orientation:
                print(f"[INFO] Stream Aspect Shift: {frame_meta['orientation']} ({frame_meta['width']}x{frame_meta['height']})")
                last_orientation = frame_meta["orientation"]

            # Asynchronous inference execution across pools
            loop = asyncio.get_event_loop()
            futures = []
            run_fast = mode in ("MILITARY_SPEC", "FAST_ONLY")
            run_broad = mode in ("MILITARY_SPEC", "BROAD_ONLY") and \
                (frame_count % ADMIN_CONFIG["broad_engine_every_n_frames"] == 0)

            if run_fast:
                futures.append(loop.run_in_executor(executor, run_fast_engine, model_fast, working_frame, min_conf))
            if run_broad:
                futures.append(loop.run_in_executor(executor, run_broad_engine, model_broad, working_frame, min_conf))

            results = await asyncio.gather(*futures) if futures else []
            raw_detections = [d for sub in results for d in sub]

            filtered = [d for d in raw_detections if (d['x2'] - d['x1']) * (d['y2'] - d['y1']) <= max_area_px]
            final_targets = non_max_suppression_fast(filtered)

            telemetry_payload = draw_and_package(frame, final_targets)
            draw_osd(frame, mode, min_conf, frame_meta)

            # Safely attempt sending WebSocket payload if connected
            if telemetry_payload and websocket:
                try:
                    payload = {
                        "type": "detection",
                        "stream_id": "drone1",
                        "frame_meta": frame_meta,
                        "targets": telemetry_payload
                    }
                    await websocket.send(json.dumps(payload))
                except Exception:
                    websocket = None  # Silently drop server pointer if dropped mid-flight

            # GUI output rendering
            display_frame = resize_for_display(frame, target_height=720)
            cv2.imshow(window_title, display_frame)
            
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break

            await asyncio.sleep(0.001)

    except Exception as e:
        print(f"[ERROR] Critical ingestion loop error: {e}")
    finally:
        print("[INFO] Shutting down RTMP thread and destroying display windows...")
        reader.stop()
        executor.shutdown(wait=False)
        cv2.destroyAllWindows()


if __name__ == "__main__":
    asyncio.run(run_pipeline())

import structlog
import logging
import sys
from config_module.settings import settings

def setup_logging():
    logging.basicConfig(
        format="%(message)s",
        stream=sys.stdout,
        level=getattr(logging, settings.LOG_LEVEL.upper(), logging.INFO),
    )

    structlog.configure(
        processors=[
            structlog.contextvars.merge_contextvars,
            structlog.processors.add_log_level,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.JSONRenderer()
        ],
        logger_factory=structlog.stdlib.LoggerFactory(),
        wrapper_class=structlog.stdlib.BoundLogger,
        cache_logger_on_first_use=True,
    )

def get_logger(module_name: str):
    return structlog.get_logger(module_name)

# message_broker/redis_producer.py
import redis.asyncio as aioredis
import json
from config_module.settings import settings
from logging_module.logger import get_logger

logger = get_logger("redis_producer")

class StreamProducer:
    def __init__(self, redis_url: str = None):
        url = redis_url or getattr(settings, 'REDIS_URL', 'redis://localhost:6379')
        self.redis = aioredis.from_url(url, decode_responses=True)

    async def publish_telemetry(self, drone_id: str, telemetry_data: dict):
        """Pushes real-time MAVLink telemetry into a Redis Stream."""
        stream_key = f"stream:telemetry:{drone_id}"
        await self.redis.xadd(
            stream_key,
            {"data": json.dumps(telemetry_data)},
            maxlen=1000,
            approximate=True
        )

    async def publish_frame_metadata(self, stream_id: str, frame_data: dict):
        """Pushes decoded video frame payload into a Redis Stream."""
        stream_key = f"stream:frames:{stream_id}"
        await self.redis.xadd(
            stream_key,
            frame_data,
            maxlen=500,
            approximate=True
        )

    async def close(self):
        await self.redis.aclose()
